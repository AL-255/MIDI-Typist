cmake_minimum_required(VERSION 3.20)
project(MIDI_Typist VERSION 0.1.0 LANGUAGES C ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(MT_BOARD huntsman_v3_pro_mini CACHE STRING "Board port in firmware/boards")
if(NOT MT_BOARD MATCHES "^[a-zA-Z0-9_]+$")
    message(FATAL_ERROR "MT_BOARD must be a board directory name")
endif()
set(MT_BOARD_DIR "${CMAKE_CURRENT_SOURCE_DIR}/firmware/boards/${MT_BOARD}")
if(NOT EXISTS "${MT_BOARD_DIR}/board.cmake")
    message(FATAL_ERROR "Unknown board: ${MT_BOARD}; provide firmware/boards/${MT_BOARD}/board.cmake")
endif()
include(cmake/build_identity.cmake)
include(firmware/app/CMakeLists.txt)
include("${MT_BOARD_DIR}/board.cmake")
# Every board target receives the same generated provenance before compiling.
get_property(mt_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(mt_target IN LISTS mt_targets)
    get_target_property(mt_type ${mt_target} TYPE)
    if(NOT mt_type STREQUAL "UTILITY" AND NOT mt_type STREQUAL "INTERFACE_LIBRARY")
        add_dependencies(${mt_target} mt_build_identity)
    endif()
endforeach()
